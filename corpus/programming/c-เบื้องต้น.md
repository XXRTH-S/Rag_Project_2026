# ภาษา C เบื้องต้น

## ภาพรวมของภาษา C

C เป็นภาษาระดับต่ำที่ใกล้เคียงกับการทำงานของฮาร์ดแวร์มากกว่าภาษาสมัยใหม่ ผู้เขียนต้องจัดการหน่วยความจำเอง จึงควบคุมได้ละเอียดและได้โปรแกรมที่เร็ว แลกกับความเสี่ยงที่จะเขียนผิดแล้วเกิดปัญหาร้ายแรง

ใช้เขียนระบบปฏิบัติการ ไดรเวอร์ ระบบฝังตัว และส่วนที่ต้องการประสิทธิภาพสูง ภาษาอื่นหลายภาษารวมถึง Python ก็เขียนตัวแปลภาษาด้วย C

## โครงสร้างโปรแกรม C

```c
#include <stdio.h>

int main(void) {
    printf("สวัสดีชาวโลก\n");
    return 0;
}
```

บรรทัด `#include` บอกให้ดึงไฟล์ส่วนหัวเข้ามาก่อนคอมไพล์ โปรแกรมเริ่มทำงานที่ฟังก์ชัน `main` และค่าที่คืนกลับบอกระบบปฏิบัติการว่าทำงานสำเร็จหรือไม่ โดยศูนย์หมายถึงสำเร็จ

## ตัวแปรและชนิดข้อมูลใน C

```c
int age = 30;
float height = 172.5f;
double precise = 3.14159265358979;
char grade = 'A';
const int MAX_USERS = 100;
```

ต้องประกาศชนิดข้อมูลก่อนใช้เสมอ ขนาดของแต่ละชนิดขึ้นกับสถาปัตยกรรมของเครื่อง ตรวจได้ด้วย `sizeof`

C ไม่มีชนิด boolean มาแต่เดิม ใช้จำนวนเต็มแทนโดยศูนย์คือเท็จและค่าอื่นคือจริง ถ้าต้องการชื่อที่อ่านง่ายให้ `#include <stdbool.h>` แล้วใช้ `bool`

## การแสดงผลและรับค่าใน C

```c
int age;
char name[50];

printf("ชื่ออะไร: ");
scanf("%49s", name);
printf("อายุเท่าไร: ");
scanf("%d", &age);

printf("สวัสดีคุณ %s อายุ %d ปี\n", name, age);
```

`%d` แทนจำนวนเต็ม `%f` แทนทศนิยม `%s` แทนข้อความ และ `%c` แทนอักขระตัวเดียว เครื่องหมาย `&` หน้าตัวแปรคือการส่งที่อยู่ของตัวแปรไปให้ `scanf` เขียนค่าลงไป

การจำกัดความยาวด้วย `%49s` สำคัญมาก ถ้าไม่จำกัด ข้อมูลที่ยาวเกินจะเขียนทับหน่วยความจำข้างเคียง เรียกว่า buffer overflow ซึ่งเป็นช่องโหว่ความปลอดภัยที่ร้ายแรง

## เงื่อนไขและการวนซ้ำใน C

```c
int score = 75;

if (score >= 80) {
    printf("A\n");
} else if (score >= 70) {
    printf("B\n");
} else {
    printf("C\n");
}

for (int i = 1; i <= 5; i++) {
    printf("%d ", i);
}

int count = 3;
while (count > 0) {
    printf("%d ", count);
    count--;
}
```

## ฟังก์ชันใน C

```c
int add(int a, int b) {
    return a + b;
}

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}
```

C ส่งค่าพารามิเตอร์แบบสำเนา การแก้ค่าในฟังก์ชันจึงไม่กระทบตัวแปรของผู้เรียก ถ้าต้องการให้แก้ได้ต้องส่งพอยน์เตอร์ไปแทน อย่างในฟังก์ชัน `swap`

## พอยน์เตอร์ใน C

พอยน์เตอร์คือตัวแปรที่เก็บที่อยู่ของข้อมูลอื่นแทนที่จะเก็บค่าโดยตรง

```c
int value = 42;
int *ptr = &value;

printf("%d\n", value);
printf("%p\n", (void *)ptr);
printf("%d\n", *ptr);

*ptr = 100;
printf("%d\n", value);
```

เครื่องหมาย `&` อ่านว่า "ที่อยู่ของ" ส่วน `*` หน้าพอยน์เตอร์อ่านว่า "ค่าที่อยู่ตรงนั้น" บรรทัดสุดท้ายพิมพ์ 100 เพราะแก้ผ่านพอยน์เตอร์ก็คือแก้ที่ตัวแปรเดิม

พอยน์เตอร์ที่ยังไม่ได้ชี้ไปที่ไหนควรตั้งเป็น `NULL` และต้องตรวจก่อนใช้ทุกครั้ง การอ่านค่าจากพอยน์เตอร์ที่ไม่ได้กำหนดค่าจะทำให้โปรแกรมพังแบบคาดเดาไม่ได้

## อาร์เรย์และสตริงใน C

```c
int numbers[5] = {10, 20, 30, 40, 50};
char name[] = "สมชาย";

for (int i = 0; i < 5; i++) {
    printf("%d ", numbers[i]);
}
```

สตริงใน C คืออาร์เรย์ของอักขระที่ปิดท้ายด้วยอักขระศูนย์ ถ้าอักขระปิดท้ายหายไป ฟังก์ชันอย่าง `printf` จะอ่านเลยขอบเขตไปเรื่อย ๆ จนเจอศูนย์ตัวถัดไปในหน่วยความจำ

C ไม่ตรวจว่าเลขตำแหน่งเกินขอบอาร์เรย์หรือไม่ การเขียนที่ `numbers[10]` จะไปทับหน่วยความจำอื่นโดยไม่มีคำเตือน

## การจองหน่วยความจำใน C

```c
#include <stdlib.h>

int *data = malloc(100 * sizeof(int));
if (data == NULL) {
    return 1;
}

data[0] = 42;
free(data);
data = NULL;
```

`malloc` จองหน่วยความจำตอนโปรแกรมทำงาน และต้องคืนด้วย `free` เสมอ ถ้าลืมคืนจะเกิด memory leak คือหน่วยความจำถูกจองไว้เรื่อย ๆ จนเต็ม

หลังเรียก `free` ควรตั้งพอยน์เตอร์เป็น `NULL` เพื่อกันการใช้พอยน์เตอร์ที่คืนไปแล้ว ซึ่งเรียกว่า dangling pointer และเป็นต้นเหตุของช่องโหว่ความปลอดภัยจำนวนมาก
