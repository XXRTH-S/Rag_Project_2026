# ภาษา Go เบื้องต้น

## ภาพรวมของภาษา Go

Go เป็นภาษาที่ออกแบบให้เรียบง่ายและคอมไพล์เร็ว มีไวยากรณ์น้อยกว่าภาษาอื่นโดยตั้งใจ เพื่อให้ทีมขนาดใหญ่เขียนโค้ดออกมาหน้าตาใกล้เคียงกัน นิยมใช้กับงานเซิร์ฟเวอร์ เครื่องมือบรรทัดคำสั่ง และระบบที่ต้องรับงานพร้อมกันจำนวนมาก

Go เป็น static typing และคอมไพล์เป็นไฟล์เดียวที่รันได้เลยโดยไม่ต้องติดตั้งอะไรเพิ่มบนเครื่องปลายทาง

## โครงสร้างโปรแกรม Go

```go
package main

import "fmt"

func main() {
    fmt.Println("สวัสดีชาวโลก")
}
```

ทุกไฟล์ต้องประกาศ package และโปรแกรมที่รันได้ต้องอยู่ใน `package main` พร้อมฟังก์ชัน `main`

Go เข้มงวดมากเรื่องของที่ไม่ได้ใช้ ตัวแปรหรือ import ที่ประกาศแล้วไม่ได้ใช้จะคอมไพล์ไม่ผ่าน ไม่ใช่แค่เตือน

## ตัวแปรและชนิดข้อมูลใน Go

```go
var name string = "สมชาย"
age := 30
height := 172.5
isActive := true
const MaxUsers = 100
```

เครื่องหมาย `:=` ประกาศตัวแปรพร้อมกำหนดค่าโดยให้ Go เดาชนิดเอง ใช้ได้เฉพาะภายในฟังก์ชัน ส่วน `var` ใช้ได้ทุกที่

ตัวแปรที่ประกาศแล้วยังไม่กำหนดค่าจะได้ค่าเริ่มต้นตามชนิด คือศูนย์สำหรับตัวเลข ข้อความว่างสำหรับ string และ false สำหรับ bool ไม่มีค่า undefined แบบภาษาอื่น

## เงื่อนไขและการวนซ้ำใน Go

```go
score := 75

if score >= 80 {
    fmt.Println("A")
} else if score >= 70 {
    fmt.Println("B")
} else {
    fmt.Println("C")
}

for i := 1; i <= 5; i++ {
    fmt.Println(i)
}

count := 3
for count > 0 {
    fmt.Println(count)
    count--
}
```

Go มีคำสั่งวนซ้ำแค่ `for` คำเดียว ไม่มี `while` แยกต่างหาก เขียน `for` ตามด้วยเงื่อนไขอย่างเดียวก็ทำงานเหมือน while และเขียน `for` เปล่า ๆ คือวนไม่รู้จบ

เงื่อนไขไม่ต้องใส่วงเล็บ แต่ปีกกาเปิดต้องอยู่บรรทัดเดียวกับ `if` เสมอ

## สไลซ์และแมปใน Go

```go
fruits := []string{"แอปเปิล", "กล้วย"}
fruits = append(fruits, "ส้ม")

for index, fruit := range fruits {
    fmt.Println(index, fruit)
}

person := map[string]int{"สมชาย": 30}
person["สมหญิง"] = 25

age, exists := person["สมศักดิ์"]
if !exists {
    fmt.Println("ไม่พบข้อมูล")
}
```

การอ่านค่าจากแมปคืนสองค่าได้ คือค่าที่เจอกับตัวบอกว่ามีคีย์นั้นจริงหรือไม่ ถ้าไม่ตรวจตัวที่สอง จะแยกไม่ออกระหว่าง "ไม่มีคีย์" กับ "มีคีย์แต่ค่าเป็นศูนย์"

## ฟังก์ชันใน Go

```go
func add(a int, b int) int {
    return a + b
}

func divide(a, b float64) (float64, error) {
    if b == 0 {
        return 0, fmt.Errorf("หารด้วยศูนย์ไม่ได้")
    }
    return a / b, nil
}
```

Go คืนค่าหลายค่าได้ ซึ่งเป็นพื้นฐานของวิธีจัดการข้อผิดพลาดในภาษานี้ ชนิดของค่าที่คืนเขียนไว้หลังรายการพารามิเตอร์

## การจัดการข้อผิดพลาดใน Go

Go ไม่มี exception แต่คืน error เป็นค่าธรรมดาแล้วให้ผู้เรียกตรวจเอง

```go
result, err := divide(10, 0)
if err != nil {
    fmt.Println("เกิดข้อผิดพลาด:", err)
    return
}
fmt.Println(result)
```

รูปแบบ `if err != nil` ปรากฏทั่วโค้ด Go จนดูซ้ำซาก แต่ข้อดีคือเห็นชัดว่าตรงไหนพลาดได้บ้าง ไม่มีข้อผิดพลาดกระโดดข้ามหลายชั้นแบบ exception

## Goroutine และแชนเนล

```go
func worker(id int, done chan string) {
    done <- fmt.Sprintf("งานที่ %d เสร็จแล้ว", id)
}

func main() {
    done := make(chan string)

    for i := 1; i <= 3; i++ {
        go worker(i, done)
    }

    for i := 1; i <= 3; i++ {
        fmt.Println(<-done)
    }
}
```

คำสั่ง `go` หน้าการเรียกฟังก์ชันทำให้ฟังก์ชันนั้นทำงานพร้อมกับโค้ดส่วนอื่น เรียกว่า goroutine ซึ่งกินทรัพยากรน้อยกว่าเธรดของระบบปฏิบัติการมาก จึงสร้างพร้อมกันเป็นหมื่นตัวได้

แชนเนลคือท่อส่งข้อมูลระหว่าง goroutine เครื่องหมาย `<-` ชี้ทิศทางของข้อมูล การอ่านจากแชนเนลจะรอจนกว่าจะมีข้อมูลส่งมา ซึ่งใช้เป็นวิธีรอให้งานเสร็จได้ในตัว
